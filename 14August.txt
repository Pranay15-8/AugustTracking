class Solution {
    public List<String> binaryTreePaths(TreeNode root) {
        if(root==null)
        return new ArrayList<String>();
        List<String> ans=new ArrayList<>();
        fun(ans,root,"");
        return ans;
    }
    public void fun(List<String> ans,TreeNode node,String str){
        str=str+node.val;
        if(node.left==null&&node.right==null){
            ans.add(str);
            return;
        }
        str+="->";
          if(node.left!=null)
          fun(ans,node.left,str);

          if(node.right!=null)
          fun(ans,node.right,str);

    }
}

class Solution {
    public int sumNumbers(TreeNode root) {
       
        if(root==null)
        return 0;
        List<String> ans=new ArrayList<>();
        fun(ans,root,"");
        int sum=0;
        for(String s:ans){
            s=s.replace("->","");
            sum+=Integer.parseInt(s);
        }
        return sum;
        
        
    }
    public void fun(List<String> ans,TreeNode node,String str){
        str=str+node.val;
        if(node.left==null&&node.right==null){
            ans.add(str);
            return;
        }
        str+="->";
          if(node.left!=null)
          fun(ans,node.left,str);

          if(node.right!=null)
          fun(ans,node.right,str);
    }
}

class Solution {
    public int sumNumbers(TreeNode root) {
       return inorder(root,0);   
        
    } 
    public int inorder(TreeNode root,int sum){
        if(root==null)
        return 0;
         
         sum=sum*10+root.val;
         if(root.left==null&&root.right==null)
         return sum;

         return inorder(root.left,sum)+inorder(root.right,sum);
        
    }
}
