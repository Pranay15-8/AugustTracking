class Solution {
    public String decodeString(String s) {
        /*IS -> Integer Stack
          SS -> String Stack*/
       Stack<Integer> IS=new Stack<>();
       Stack<String> SS=new Stack<>();
       StringBuilder sb=new StringBuilder();
       int n=0;
       for(char ch:s.toCharArray()){
        if(Character.isDigit(ch)){
            n=n*10+(ch-'0');
        }
        else if(ch=='['){
            IS.push(n);
            n=0;
            SS.push(sb.toString());
            sb=new StringBuilder();
        }
        else if(ch==']'){
            StringBuilder temp=new StringBuilder(SS.pop());
            int x=IS.pop();
            for(int i=0;i<x;i++){
                temp.append(sb);
            }
            sb=temp;
        }else{
            sb.append(ch);
        }

       }
       return sb.toString(); 
    }
}

class Solution {
    public int calPoints(String[] operations) {
        Stack<Integer> st=new Stack<>();
        for(int i=0;i<operations.length;i++){
             String s=operations[i];
             if(!s.equals("C")&&!s.equals("D")&&!s.equals("+")){
                st.push(Integer.parseInt(s));
             }
             else if(s.equals("D")){
                int x=0;
                if(!st.isEmpty())
                 x=st.peek();
                 st.push(2*x);
             }
             else if(s.equals("+")){
                int a=st.pop();
               
                 int b=st.peek();
                 st.push(a);
                st.push(a+b);
             }
             else if(s.equals("C")){
                st.pop();
             }
        }
        int sum=0;
        while(!st.isEmpty())
        sum+=st.pop();
        return sum;
    }
}
