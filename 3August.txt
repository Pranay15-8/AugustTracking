class Solution {
    public boolean isPalindrome(ListNode head) {
        if(head.next==null)
        return true;
       ListNode slow=head;
      ListNode fast=head;
       while(fast.next!=null&&fast.next.next!=null){
          slow=slow.next;
          fast=fast.next.next;
       }
       ListNode newHead=reverse(slow.next);
       ListNode first=head;
       ListNode second=newHead;
       while(second!=null){
         if(first.val!=second.val){
            reverse(newHead);
            return false;
         }
         first=first.next;
         second=second.next;
       }
       reverse(newHead);
       return true;

    }
    public ListNode reverse(ListNode head){
        if(head.next==null)
        return head;

        ListNode newHead=reverse(head.next);
        ListNode front=head.next;
        front.next=head;
        head.next=null;
        return newHead;
    }
}

//Finding the length of the loop in the linkedlist
class Solution {
    public int lengthOfLoop(Node head) {
        // code here
        Node slow=head;
        Node fast=head;
        while(fast!=null&&fast.next!=null){
            slow=slow.next;
            fast=fast.next.next;
            if(fast==slow){
               return length(slow,fast);
            }
        }
        return 0;
    }
    public int length(Node slow,Node fast){
       int c=0;
        do{
            c++;
            slow=slow.next;
        }while(slow!=fast);
        return c;
    }
}
