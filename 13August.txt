public List<Integer> preOder(TreeNode root){
   List<Integer> list=new ArrayList<>();
        if(root==null)
        return list;
        Stack<TreeNode> st=new Stack<>();
        st.push(root);
        while(!st.isEmpty()){
            TreeNode n=st.pop();
            list.add(new Integer(n.val));
            if(n.right!=null){
                st.push(n.right);
            }
            if(n.left!=null){
                st.push(n.left);
            }
        }
        return list;
}

          List<Integer> list=new ArrayList<>();

       while(true){
        if(n!=null){
            st.push(n);
            n=n.left;
        }
        else{
            if(st.isEmpty())
            break;

            n=st.pop();
            list.add(new Integer(n.val));

        }
       }

    }
