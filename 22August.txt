
class Solution {
    public boolean checkDivisibility(int n) {
        int x=n;
        int s=0,p=1;
        while(n>0){
            int r=n%10;
            s+=r;
            p*=r;
            n/=10;
        }
        return (x%(s+p))==0;
    }
}
